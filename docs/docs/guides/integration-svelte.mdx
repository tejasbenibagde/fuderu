# Svelte Integration Guide

This guide covers best practices for integrating Fuderu with Svelte applications, including stores, component patterns, and performance optimization.

## Basic Setup

### Installation

```bash
npm install fuderu
# or
yarn add fuderu
```

### Basic Usage

```svelte
<script>
  import { onMount, onDestroy } from 'svelte';
  import { Canvas } from 'fuderu';

  let canvas;
  let painter = null;

  onMount(() => {
    if (!canvas) return;

    painter = new Canvas({
      canvas,
      document: {
        width: 800,
        height: 600
      }
    });
  });

  onDestroy(() => {
    if (painter) {
      painter.destroy();
      painter = null;
    }
  });
</script>

<canvas bind:this={canvas} width="800" height="600" />
```

## Stores

### Canvas Store

Create a reusable store for canvas management:

```js
// stores/canvas.js
import { writable } from "svelte/store";
import { Canvas } from "fuderu";

export function createCanvasStore(options = {}) {
  const { subscribe, set, update } = writable({
    canvas: null,
    painter: null,
    width: options.width ?? 800,
    height: options.height ?? 600,
    brushConfig: options.brushConfig ?? {},
  });

  let painterInstance = null;

  return {
    subscribe,
    setCanvasElement: (canvasElement) => {
      update((state) => ({ ...state, canvas: canvasElement }));
      initialize(canvasElement);
    },
    setSize: (width, height) => {
      update((state) => ({
        ...state,
        width,
        height,
      }));
      resizeCanvas(width, height);
    },
    updateBrushConfig: (newConfig) => {
      update((state) => ({
        ...state,
        brushConfig: { ...state.brushConfig, ...newConfig },
      }));
      applyBrushConfig(newConfig);
    },
    getPainter: () => painterInstance,
    destroy: () => {
      if (painterInstance) {
        painterInstance.destroy();
        painterInstance = null;
      }
      update((state) => ({ ...state, painter: null }));
    },
  };

  function initialize(canvasElement) {
    if (!canvasElement) return;

    // Clean up existing instance
    if (painterInstance) {
      painterInstance.destroy();
    }

    // Create new instance
    const { width, height, brushConfig } = getStoreValue();
    painterInstance = new Canvas({
      canvas: canvasElement,
      document: { width, height },
      brush: brushConfig,
    });

    update((state) => ({ ...state, painter: painterInstance }));
  }

  function resizeCanvas(width, height) {
    if (painterInstance) {
      painterInstance.setDocumentSize(width, height);
    }
  }

  function applyBrushConfig(newConfig) {
    if (painterInstance) {
      Object.assign(painterInstance.brush.config, newConfig);
    }
  }

  // Helper to get current store value
  function getStoreValue() {
    let value;
    const unsub = subscribe((val) => {
      value = val;
    })();
    unsub();
    return value;
  }
}

// Usage in components:
// import { canvasStore } from './stores/canvas';
// canvasStore.setCanvasElement(canvas);
```

## Component Patterns

### DrawingCanvas Component

```svelte
<!-- DrawingCanvas.svelte -->
<script>
  import { onMount, onDestroy } from 'svelte';
  import { Canvas } from 'fuderu';
  import { brushStore } from './stores/brushStore';

  export let width = 800;
  let height = 600;
  let canvas;
  let painter = null;

  // Subscribe to brush store
  let brushUnsubscribe;

  onMount(() => {
    brushUnsubscribe = brushStore.subscribe((brushConfig) => {
      if (painter) {
        Object.assign(painter.brush.config, brushConfig);
      }
    });

    if (!canvas) return;

    painter = new Canvas({
      canvas,
      document: { width, height }
    });
  });

  onDestroy(() => {
    if (brushUnsubscribe) brushUnsubscribe();
    if (painter) {
      painter.destroy();
      painter = null;
    }
  });

  // Handle resize
  function handleResize() {
    if (painter) {
      // Update dimensions from canvas size
      const rect = canvas.getBoundingClientRect();
      height = rect.height;
      // Width is typically fixed or controlled externally
      painter.setDocumentSize(rect.width, rect.height);
    }
  }
</script>

<canvas
  bind:this={canvas}
  width={width}
  height={height}
  on:resize={handleResize}
/>
```

### Animation Loop with Svelte's Ticker

```svelte
<script>
  import { onMount, onDestroy } from 'svelte';
  import { Canvas } from 'fuderu';
  import { tick } from 'svelte';

  let canvas;
  let painter = null;
  let animationFrame = null;
  let needsRender = false;

  onMount(() => {
    if (!canvas) return;

    painter = new Canvas({
      canvas,
      // ... options
    });

    startRenderLoop();
  });

  onDestroy(() => {
    stopRenderLoop();
    if (painter) {
      painter.destroy();
    }
  });

  function startRenderLoop() {
    function loop() {
      if (needsRender && painter) {
        painter.render();
        needsRender = false;
      }
      animationFrame = requestAnimationFrame(loop);
    }

    loop();
  }

  function stopRenderLoop() {
    if (animationFrame) {
      cancelAnimationFrame(animationFrame);
      animationFrame = null;
    }
  }

  // Function to request a render
  function requestRender() {
    needsRender = true;
  }

  // Alternative: Use Svelte's upcoming run loop (experimental)
```

## Performance Optimization

### Request Animation Frame Wrapper

```js
// utils/raf.js
let frameCallback = null;
let frameRequest = null;
let lastTime = 0;

export function initRenderLoop(callback) {
  frameCallback = callback;

  function loop(timestamp) {
    const deltaTime = timestamp - lastTime;
    lastTime = timestamp;

    if (frameCallback) {
      frameCallback(deltaTime);
    }

    frameRequest = requestAnimationFrame(loop);
  }

  frameRequest = requestAnimationFrame(loop);

  return () => {
    if (frameRequest) {
      cancelAnimationFrame(frameRequest);
      frameRequest = null;
    }
    frameCallback = null;
  };
}
```

### Virtualized Canvas for Large Documents

```svelte
<!-- VirtualCanvas.svelte -->
<script>
  import { onMount, onDestroy } from 'svelte';
  import { Canvas } from 'fuderu';

  export let virtualWidth = 4000;
  export let virtualHeight = 3000;
  export let viewportWidth = 800;
  export let viewportHeight = 600;

  let container;
  let canvas;
  let painter = null;
  let offset = { x: 0, y: 0 }; // Top-left of viewport in virtual space

  let animationFrame = null;
  let needsRender = false;

  // Create canvas at viewport size, not virtual size
  onMount(() => {
    if (!container || !canvas) return;

    painter = new Canvas({
      canvas,
      document: { width: viewportWidth, height: viewportHeight }
    });

    startRenderLoop();
  });

  onDestroy(() => {
    stopRenderLoop();
    if (painter) {
      painter.destroy();
    }
  });

  function startRenderLoop() {
    function loop() {
      if (needsRender && painter) {
        // Save current state
        const savedTransform = painter.brush.context.getTransform();

        // Translate to show correct portion of virtual canvas
        painter.brush.context.save();
        painter.brush.context.translate(-offset.x, -offset.y);

        // Render only the visible portion
        painter.render();

        // Restore state
        painter.brush.context.restore();
        painter.brush.context.setTransform(savedTransform);

        needsRender = false;
      }
      animationFrame = requestAnimationFrame(loop);
    }

    loop();
  }

  function stopRenderLoop() {
    if (animationFrame) {
      cancelAnimationFrame(animationFrame);
      animationFrame = null;
    }
  }

  function requestRender() {
    needsRender = true;
  }

  // Pan handling
  function handleWheel(event) {
    event.preventDefault();

    // Convert wheel delta to pan movement
    const panSpeed = 0.5; // pixels per delta unit
    offset.x = Math.max(0, Math.min(virtualWidth - viewportWidth,
      offset.x + event.deltaX * panSpeed));
    offset.y = Math.max(0, Math.min(virtualHeight - viewportHeight,
      offset.y + event.deltaY * panSpeed));

    requestRender();
  }

  // Click-drag panning
  let isPanning = false;
  let lastPanPoint = null;

  function handleMouseDown(event) {
    isPanning = true;
    lastPanPoint = { x: event.clientX, y: event.clientY };
  }

  function handleMouseMove(event) {
    if (!isPanning || !lastPanPoint) return;

    const dx = event.clientX - lastPanPoint.x;
    const dy = event.clientY - lastPanPoint.y;

    offset.x = Math.max(0, Math.min(virtualWidth - viewportWidth,
      offset.x - dx));
    offset.y = Math.max(0, Math.min(virtualHeight - viewportHeight,
      offset.y - dy));

    lastPanPoint = { x: event.clientX, y: event.clientY };
    requestRender();
  }

  function handleMouseUp() {
    isPanning = false;
    lastPanPoint = null;
  }

  function handleMouseLeave() {
    isPanning = false;
    lastPanPoint = null;
  }
</script>

<div
  bind:this={container}
  class="canvas-container"
  on:wheel={handleWheel}
  on:mousedown={handleMouseDown}
  on:mousemove={handleMouseMove}
  on:mouseup={handleMouseUp}
  on:mouseleave={handleMouseLeave}
  style="position: relative; overflow: hidden;"
>
  <canvas
    bind:this={canvas}
    width={viewportWidth}
    height={viewportHeight}
    style="position: absolute; top: 0; left: 0;"
  />

  {/* Optional: show viewport indicator */}
  <div
    class="viewport-indicator"
    style="
      position: absolute;
      border: 2px dashed #007acc;
      pointer-events: none;
    "
  />
</div>

<style>

    width: 100%;
    height: 100%;
    touch-action: none; /* Important for panning */
  }

  .viewport-indicator {
    /* Will be positioned by JS to show current viewport */
  }
</style>
```

### Event Handling and Delegation

```svelte
<script>
  import { onMount, onDestroy } from 'svelte';
  import { Canvas } from 'fuderu';

  let canvas;
  let painter = null;

  // Event handler reuse to prevent creating new functions on each render
  function handlePointerDown(event) {
    // ... implementation
  }

  function handlePointerMove(event) {
    // ... implementation
  }

  function handlePointerUp(event) {
    // ... implementation
  }

  onMount(() => {
    if (!canvas) return;

    painter = new Canvas({
      canvas,
      // ... options
    });

    // Use captured event handlers
    canvas.addEventListener('pointerdown', handlePointerDown);
    canvas.addEventListener('pointermove', handlePointerMove);
    canvas.addEventListener('pointerup', handlePointerUp);
  });

  onDestroy(() => {
    if (!canvas) return;

    // Remove the same function references
    canvas.removeEventListener('pointerdown', handlePointerDown);
    canvas.removeEventListener('pointermove', handlePointerMove);
    canvas.removeEventListener('pointerup', handlePointerUp);

    if (painter) {
      painter.destroy();
    }
  });
</script>
```

### Lifecycle Awareness with Stores

```js
import { writable } from "svelte/store";

export function createLifecycleStore() {
  const { subscribe, set } = writable({
    isMounted: false,
```
